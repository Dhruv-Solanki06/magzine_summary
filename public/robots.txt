# Article and magazine pages stay fully crawlable — they are the content worth
# indexing. What is blocked below is the filter/search surface: those URLs are a
# combinatorial space (search x language x year x tags x authors x sort x page)
# that resolves to the same articles, so crawling them adds no index value and
# generates unbounded database reads.

User-agent: *
Disallow: /api/
Disallow: /*?search=
Disallow: /*?*search=
Disallow: /*?*tags=
Disallow: /*?*authors=
Disallow: /*?*sort=
Disallow: /*?*language=
Disallow: /*?*yearStart=
Disallow: /*?*yearEnd=
Disallow: /*?*page=
Disallow: /*?*magazineId=
Disallow: /account
Disallow: /bookmarks
Disallow: /favorites
Disallow: /tracker
Disallow: /profile/edit
Disallow: /profile$
Disallow: /login
Disallow: /signup
Disallow: /forgot-password
Disallow: /reset-password
Disallow: /auth/

# Aggressive crawlers: same rules, plus a crawl delay. A crawler obeys only the
# most specific group that names it and ignores `*` entirely, so each group
# must repeat the full Disallow list — a group with just Crawl-delay would let
# these bots crawl everything above. Google ignores Crawl-delay (tune it in
# Search Console instead); Bing and Yandex honour it.
